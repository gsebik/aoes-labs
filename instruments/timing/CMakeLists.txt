cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico2_w CACHE STRING "AoES reference board")
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(aoes_timing C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(aoes_timing main.c)
target_include_directories(aoes_timing PRIVATE ../../common)
target_link_libraries(aoes_timing pico_stdlib hardware_clocks hardware_adc m)
pico_enable_stdio_uart(aoes_timing 1)
pico_enable_stdio_usb(aoes_timing 0)
pico_add_extra_outputs(aoes_timing)
