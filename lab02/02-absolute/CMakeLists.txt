cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico2_w CACHE STRING "Reference board")
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(aoes_lab02_absolute C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(aoes_lab02_absolute main.c)
target_link_libraries(aoes_lab02_absolute pico_stdlib)
target_compile_options(aoes_lab02_absolute PRIVATE -Wall -Wextra)
pico_enable_stdio_uart(aoes_lab02_absolute 1)
pico_enable_stdio_usb(aoes_lab02_absolute 0)
pico_add_extra_outputs(aoes_lab02_absolute)
